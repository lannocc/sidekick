import os
import zipfile
import google.generativeai as genai
from google.ai import genai as legacy_genai  # For older models like Gemini 1.0 Pro

def zip_project(project_dir, output_zip_filename):
    """Zips the contents of a directory into a zip file.

    Args:
        project_dir: The path to the directory containing the project.
        output_zip_filename: The name of the zip file to create.
    """
    try:
        with zipfile.ZipFile(output_zip_filename, 'w', zipfile.ZIP_DEFLATED) as zipf:
            for root, _, files in os.walk(project_dir):
                for file in files:
                    file_path = os.path.join(root, file)
                    relative_path = os.path.relpath(file_path, project_dir)
                    zipf.write(file_path, relative_path)
        print(f"Successfully zipped project to {output_zip_filename}")
    except Exception as e:
        print(f"Error zipping project: {e}")
        return None
    return output_zip_filename

def submit_project_to_gemini(zip_file_path, prompt, api_key, model_name="gemini-1.5-pro-latest", include_zip_in_prompt=True):  # More recent model
    """Submits a zipped Python project to Gemini for analysis.

    Args:
        zip_file_path: The path to the zipped project file.
        prompt: The prompt for Gemini (e.g., "Analyze this code for potential bugs.").
        api_key: Your Google Gemini API key.
        model_name: The name of the Gemini model to use (e.g., "gemini-1.5-pro-latest", "gemini-pro").
        include_zip_in_prompt: Whether to include the zip file contents directly in the prompt (for smaller zips).
                            If False, assumes the model can handle larger context windows or you're using a Vision model (with limitations on project structure).

    Returns:
        The response from Gemini, or None if there was an error.
    """

    try:
        genai.configure(api_key=api_key)
        model = genai.GenerativeModel(model_name)

        if include_zip_in_prompt:
            # Read the zip file and include its content in the prompt
            with open(zip_file_path, "rb") as f:
                zip_content = f.read().decode("latin-1", errors="ignore")  # Decode as latin-1 to handle binary data
                full_prompt = f"{prompt}\n\nProject Zip Contents:\n```\n{zip_content}\n```"
        else:
            full_prompt = prompt

        response = model.generate_content(full_prompt)  # Just using the prompt if zip contents aren't included

        print("Gemini Response:")
        print(response.text)
        return response

    except Exception as e:
        print(f"Error submitting project to Gemini: {e}")
        return None

def legacy_submit_project_to_gemini(zip_file_path, prompt, api_key): # Use for Gemini 1.0 Pro models
    """Submits a zipped Python project to Gemini 1.0 Pro (using older API).

    This function uses the older `legacy_genai` library to support older Gemini models

    This method reads the zip file and includes its content in the prompt.
    """
    try:
        legacy_genai.configure(api_key=api_key)
        model = legacy_genai.GenerativeModel('gemini-pro') # Hardcoded for 1.0 Pro

            zip_content = f.read().decode("latin-1", errors="ignore")  # Decode as latin-1 to handle binary data
            full_prompt = f"{prompt}\n\nProject Zip Contents:\n```\n{zip_content}\n```"

        response = model.generate_content(full_prompt)  # Just using the prompt if zip contents aren't included

        print("Gemini Response:")
        print(response.text)
        return response

    except Exception as e:
        print(f"Error submitting project to Gemini (legacy): {e}")
        return None
